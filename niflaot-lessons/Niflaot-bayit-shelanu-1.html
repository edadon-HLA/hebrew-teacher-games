תַּפְזֹרֶת — מוֹצְאִים אֶת הַמִּלִּים

הַמִּלִּים כְּתוּבוֹת מִיָּמִין לִשְׂמֹאל אוֹ מִלְמַעְלָה לְמַטָּה. עַל הַמָּסָךְ — לוֹחֲצִים עַל מִלָּה בָּרְשִׁימָה כְּדֵי לְסַמֵּן אוֹתָהּ בַּתַּפְזֹרֶת.

📝 מהלך קצר

מְחַלְּקִים דַּף לְכָל תַּלְמִיד כְּשֶׁהֵם נִכְנָסִים. רַק מִלִּים מֻכָּרוֹת. מַקִּיפִים לְבַד, בְּשֶׁקֶט. בַּסּוֹף תַּלְמִיד אוֹמֵר מִלָּה, הַמּוֹרֶה לוֹחֵץ עָלֶיהָ וְכָל הַכִּתָּה קוֹרֵאת יַחַד.

2 חִמּוּם · 3–5 דַּקּוֹת

אֵיזֶה צֶבַע?

לוֹחֲצִים עַל תְּמוּנָה ⬅ הַכִּתָּה מַרִימָה כַּרְטִיס צֶבַע (3-2-1 הָרִימוּ!) ⬅ הַמּוֹרֶה לוֹחֵץ עַל הַצֶּבַע הַנָּכוֹן ⬅ אוֹמְרִים אֶת הַמִּשְׁפָּט יַחַד.

🖨️ כַּרְטִיסֵי צְבָעִים + הוּא / הִיא (PDF)
📝 מהלך קצר

חִמּוּם קַל עִם צְבָעִים מֻכָּרִים, בִּתְמוּנוֹת שֶׁל דְּבָרִים בַּבַּיִת. לְכָל תַּלְמִיד כַּרְטִיסֵי צֶבַע (מֵהַשִּׁעוּר "מִי מְבַשֵּׁל"). אַחֲרֵי הַגִּלּוּי אוֹמְרִים יַחַד: "הַסַּפָּה אֲדֻמָּה", "הַשָּׁטִיחַ כָּחֹל"...

3 הַסְבֵּר · 10 דַּקּוֹת
⭐ לְפִי שִׁקּוּל הַמּוֹרֶה — חֲזָרָה קְצָרָה: מִי מְבַשֵּׁל וּמִי מְסַדֵּר? (כַּרְטִיסֵי הוּא / הִיא)

מָה אֲנַחְנוּ כְּבָר יוֹדְעִים?

✨ נִפְלָאוֹת: בַּיִת חָדָשׁ — מָסָךְ 3 (מָה יֵשׁ בַּחֲדָרִים בַּבָּתִּים שֶׁלָּנוּ?)

הַמּוֹרֶה גּוֹרֵר וּמְדַבֵּר: "בַּסָּלוֹן שֶׁלִּי יֵשׁ סַפָּה, שָׁטִיחַ וּמְנוֹרָה. בַּמִּטְבָּח שֶׁלִּי יֵשׁ מְקָרֵר וְתַנּוּר." אַחַר כָּךְ תַּלְמִיד גּוֹרֵר וְאוֹמֵר מִשְׁפָּט אֶחָד.

📝 מהלך קצר

זֶה טַבְלַת הַגְּרִירָה מִ־22/9. 3 דַּקּוֹת בְּעֵרֶךְ. הַקִּשּׁוּר הוּא לַפְּרִיט "בַּיִת חָדָשׁ" בְּאֲתַר נִפְלָאוֹת — פּוֹתְחִים אֶת מָסָךְ 3.

חָדָשׁ: מָה יֵשׁ בַּחֲדָרִים?

לוֹחֲצִים עַל דָּבָר בַּתְּמוּנָה אוֹ עַל מִלָּה ⬅ רוֹאִים וְשׁוֹמְעִים אֶת הַמִּלָּה וְאֶת הַמִּשְׁפָּט. ⭐ = מִלָּה חֲדָשָׁה (לוֹחֲצִים עָלֶיהָ קֹדֶם). הַכִּתָּה חוֹזֶרֶת פַּעֲמַיִם.

אֵיפֹה יֵשׁ ___? ⬅ בַּ___.
📝 מהלך קצר

הַתְּמוּנָה הִיא הַבַּיִת מִמִּשְׂחַק הַמַּחֲבוֹאִים (הַכִּתָּה מַכִּירָה אוֹתוֹ), עִם הַחֲפָצִים הַחֲדָשִׁים. מִלִּים בְּלִי מָקוֹם בַּתְּמוּנָה (כִּיּוֹר, סִיר, מַזְלֵג, סַבּוֹן) — לוֹחֲצִים עַל הַמִּלָּה לְמַטָּה וְרוֹאִים תְּמוּנָה. לְכָל מִלָּה מֻכֶּרֶת שׁוֹאֲלִים קֹדֶם "מָה זֶה?" וְהַכִּתָּה עוֹנָה, וְרַק אַחַר כָּךְ לוֹחֲצִים. בְּסוֹף כָּל לַשּׁוֹנִית — "בְּדִיקָה מַצְחִיקָה".

תַּרְבּוּת: בְּהַרְבֵּה דִּירוֹת בְּיִשְׂרָאֵל הַמִּקְלַחַת וְהַשֵּׁרוּתִים הֵם שְׁנֵי חֲדָרִים קְטַנִּים, וּכִמְעַט בְּכָל בַּיִת יֵשׁ מַזְגָן, כִּי בַּקַּיִץ חַם.

4 עוֹשִׂים · 15–20 דַּקּוֹת

א · מַחֲבוֹאִים — גִּרְסָה מֻרְחֶבֶת

🏠

מַחֲבוֹאִים ✨ גִּרְסָה מֻרְחֶבֶת

הַמְּקוֹמוֹת הַחֲדָשִׁים: מִיקְרוֹגַל, קֻמְקוּם, כֻּרְסָה, שַׁלָּט, מַזְגָן...

◀

קַל (2 סִבּוּבִים): הָרֶמֶז אוֹמֵר אֶת הַחֶדֶר ⬅ מַרִימִים כַּרְטִיס חֶדֶר (3-2-1 הָרִימוּ!) ⬅ תַּלְמִיד לוֹחֵץ. קָשֶׁה (2 סִבּוּבִים): "לְיַד הַמִּיקְרוֹגַל", "מֵאֲחוֹרֵי הַכֻּרְסָה" ⬅ מַרִימִים אֶת הַחֶדֶר שֶׁל הַדָּבָר ⬅ תַּלְמִיד מוֹצֵא.

🖨️ כַּרְטִיסֵי חֲדָרִים (PDF)
📝 מהלך קצר

הַמּוֹרֶה מַקְלִיד שֵׁם וּבוֹחֵר בֵּן / בַּת. מְשַׂחֲקִים בְּסָלוֹן, מִטְבָּח וְחֶדֶר רַחֲצָה. סִבּוּב בְּחֶדֶר שֵׁנָה אוֹ בַּגִּנָּה — חֲזָרָה קְצָרָה (הֵם יַחְזְרוּ בְּשִׁעוּר 4).

ב · אֵיפֹה יֵשׁ ___?

הַכִּתָּה מַרִימָה כַּרְטִיס חֶדֶר (3-2-1 הָרִימוּ!) ⬅ הַמּוֹרֶה לוֹחֵץ עַל הַחֶדֶר ⬅ אוֹמְרִים אֶת הַמִּשְׁפָּט.

📝 מהלך קצר

רָמָה קַלָּה: תְּמוּנָה + מִלָּה, מַרִימִים כַּרְטִיס תְּמוּנָה. רָמָה קָשָׁה: רַק מִלָּה, מַרִימִים כַּרְטִיס מִלָּה. מַזְגָן — גַּם חֶדֶר שֵׁנָה נָכוֹן.

ג · יֵשׁ אוֹ אֵין?

קוֹרְאִים מִשְׁפָּט ⬅ מַרִימִים ✔ אוֹ ✘ (3-2-1 הָרִימוּ!) ⬅ הַמּוֹרֶה לוֹחֵץ. אַחֲרֵי ✘ מְתַקְּנִים בְּקוֹל.

זִכָּרוֹן ✨ גִּרְסָה מֻרְחֶבֶת: חֲפָצִים בַּבַּיִת

8 זוּגוֹת · שְׁתֵּי קְבוּצוֹת

◀

שְׁתֵּי קְבוּצוֹת. תַּלְמִיד הוֹפֵךְ שְׁנֵי קְלָפִים, וְהַקְּבוּצָה קוֹרֵאת אֶת הַמִּלָּה.

🗣️ מְדַבְּרִים עַל הַבַּיִת שֶׁלִּי

בְּזוּגוֹת, פָּנִים מוּל פָּנִים. א שׁוֹאֵל וּ־ב עוֹנֶה, וּבַצִּלְצוּל מִתְחַלְּפִים.

📝 מהלך קצר

סִבּוּב 1 — הַמִּטְבָּח (2 דַּקּוֹת), סִבּוּב 2 — הַחֶדֶר שֶׁלִּי (2 דַּקּוֹת). הַמּוֹרֶה מִסְתּוֹבֵב וּמַקְשִׁיב. בַּסּוֹף 2-3 תַּלְמִידִים מְסַפְּרִים עַל הַמִּטְבָּח אוֹ הַחֶדֶר שֶׁל בֶּן הַזּוּג ("בַּמִּטְבָּח שֶׁל דָּנָה יֵשׁ מִיקְרוֹגַל."), וְהַכִּתָּה מְגִיבָה "גַּם לִי!" / "לִי אֵין!".

תְּמִיכָה (קְבוּצָה 1): מַצְבִּיעִים עַל הַתְּמוּנוֹת וְאוֹמְרִים 2 דְּבָרִים. אֶתְגָּר (קְבוּצָה 3): 3 דְּבָרִים + "אֵין": "בַּחֶדֶר שֶׁלִּי יֵשׁ מִטָּה, אָרוֹן וּמְנוֹרָה, אֲבָל אֵין לִי טֶלֶוִיזְיָה." הַשְּׁאֵלוֹת בְּזָכָר; כָּל אֶחָד עוֹנֶה בְּמִין שֶׁלּוֹ.
