בַּמָּסָךְ: תְּמוּנָה שֶׁל בַּיִת וְשֶׁל בִּנְיָן. הַמּוֹרֶה מַתְחִיל "בַּבַּיִת שֶׁלִּי יֵשׁ..." עִם מַחְסַן הַמִּלִּים, וּכַמָּה תַּלְמִידִים מַמְשִׁיכִים — דֻּגְמָה לִפְנֵי הַמַּצָּגוֹת.

📝 מהלך קצר

הַמּוֹרֶה אוֹמֵר אֶת הַטֶּקְסְט בְּקוֹל (אֶפְשָׁר לְהַחְלִיף אֶת הַתְּמוּנָה בִּתְמוּנָה שֶׁל הַבַּיִת שֶׁלּוֹ). הַכַּפְתּוֹר "מוֹרָה" מַחְלִיף לִנְקֵבָה. תַּרְבּוּת: בְּיִשְׂרָאֵל מַזְמִינִים שָׁכֵן אוֹ חָבֵר חָדָשׁ: "בּוֹא, תִּרְאֶה אֶת הַבַּיִת!" וּמַרְאִים חֶדֶר אַחֲרֵי חֶדֶר.

📝 מהלך קצר

הַמּוֹרֶה מַרְאֶה אֶת הַמִּסְגֶּרֶת. הַכִּתָּה מוֹצֵאת כָּל פְּרִיט בָּרְשִׁימָה בַּדֻּגְמָה שֶׁל הַמּוֹרֶה וּמַרִימָה יָד. לוֹחֲצִים עַל פְּרִיט בָּרְשִׁימָה כְּדֵי לְסַמֵּן ✔.

א · רֵאָיוֹן עִם חָבֵר חָדָשׁ (6 דַּקּוֹת)

כַּרְטִיס הָרֵאָיוֹן מִשִּׁעוּר 4 — עִם חָבֵר שֶׁעוֹד לֹא רִאְיַנְתֶּם. שׁוֹאֲלִים אֶת 7 הַשְּׁאֵלוֹת וְכוֹתְבִים בָּעַמּוּדָה "הֶחָבֵר שֶׁלִּי". (טַיְמֶר 6′ לְמַעְלָה.)

🖨️ כַּרְטִיס רֵאָיוֹן (PDF)

ב · מְצַיְּרִים: הַתָּכְנִית שֶׁל הַבַּיִת שֶׁלִּי (5 דַּקּוֹת)

קֻפְסָה לְכָל חֶדֶר, שֵׁם הַחֶדֶר בְּעִבְרִית (מִטְבָּח, סָלוֹן, חֶדֶר שֵׁנָה, חֶדֶר רַחֲצָה, מִרְפֶּסֶת), 2 דְּבָרִים בְּכָל חֶדֶר. מִי שֶׁגָּר בְּבִנְיָן — מוֹסִיף קוֹמָה וּמַעֲלִית.

🖨️ דַּף תָּכְנִית (PDF · לֹא חוֹבָה)

ג · חֲזָרָה בְּזוּגוֹת (3 דַּקּוֹת)

כָּל אֶחָד אוֹמֵר אֶת הַמַּצֶּגֶת פַּעַם אַחַת לְבֶן הַזּוּג. בֶּן הַזּוּג מַצְבִּיעַ עַל הָרְשִׁימָה וְנוֹתֵן טִיפּ אֶחָד: "חָסֵר: ___".

ד · הַמַּצָּגוֹת (10 דַּקּוֹת)

קְבוּצוֹת שֶׁל 4: כָּל אֶחָד 30-60 שְׁנִיּוֹת עִם הַצִּיּוּר (טַיְמֶר אֶחָד, "3-2-1 מַתְחִילִים!"). אַחֲרֵי כָּל מַצֶּגֶת — כָּל מַקְשִׁיב שׁוֹאֵל שְׁאֵלָה אַחַת ("יֵשׁ לְךָ מַזְגָן בַּחֶדֶר?" / "בְּאֵיזוֹ קוֹמָה?" / "מִי מְבַשֵּׁל?"). אַחַר כָּךְ 3-4 מִתְנַדְּבִים מוּל הַכִּתָּה; הַמּוֹרֶה שׁוֹאֵל 2 שְׁאֵלוֹת ✔ / ✘, וְכֻלָּם: "כָּל הַכָּבוֹד!"

📝 מהלך קצר

הַמּוֹרֶה מִסְתּוֹבֵב עִם רְשִׁימַת הַבְּדִיקָה: נְקֻדָּה לְכָל פְּרִיט (5) + עָנָה עַל שְׁאֵלָה. תְּמִיכָה (קְבוּצָה 1): 3 פְּרִיטִים, קוֹרְאִים מֵהַמִּסְגֶּרֶת. אֶתְגָּר (קְבוּצָה 3): מַצִּיגִים אֶת הֶחָבֵר (הוּא / הִיא), 2 פְּעָלִים בְּכָל חֶדֶר וּמִשְׁפָּט עִם "אֵין".
